/**
 * 防災のまとめ
 * 最新災害情報｜SNSシェア導線 v1.4.2
 *
 * 配置:
 * - data-bousai-slot="post-current-actions" がある場合は、そのslot内
 * - slot内では防災チェックCTAの後ろ
 * - slotがまだ無い旧実装では「現在発表されている警報・災害情報」の直後へfallback
 *
 * 対象:
 * - /disaster-info/
 * - /disaster-info/{prefecture}/
 *
 * ボタン順:
 * - LINE / X / Threads / Bluesky / リンクをコピー
 *
 * 方針:
 * - 既存の災害情報取得・地域判定・キャッシュ・保存HTMLには触れない。
 * - 最新災害情報layout ownerが提供する安定slotを優先し、root-level配置を奪い合わない。
 * - slotが存在しない場合だけ従来配置へfallbackし、後方互換を維持する。
 * - 共有URLは canonical を優先し、都道府県ページの正規URLを共有する。
 * - 外部JSライブラリは追加しない。
 */

function bousai_matome_disaster_share_is_target_page() {
    if ( function_exists( 'bousai_matome_is_disaster_info_page' ) ) {
        return (bool) bousai_matome_is_disaster_info_page();
    }

    if ( is_page( 'disaster-info' ) ) {
        return true;
    }

    $queried = get_queried_object();
    return $queried instanceof WP_Post && $queried->post_name === 'disaster-info';
}

    <script id="bousai-disaster-share-script">
    (function(){
        'use strict';

        var SHARE_ID = 'bousai-disaster-share';

        function normalizeText(value){
            return (value || '').replace(/\s+/g, ' ').trim();
        }

        function findCurrentSection(root){
            var sections = root.querySelectorAll('.bousai-source-section');

            for (var i = 0; i < sections.length; i++) {
                var title = sections[i].querySelector('.bousai-source-title');
                if (title && normalizeText(title.textContent) === '現在発表されている警報・災害情報') {
                    return sections[i];
                }
            }

            return null;
        }

        function findPostCurrentActionsSlot(root){
            return root.querySelector(':scope > [data-bousai-slot="post-current-actions"]');
        }

        function getRegion(root){
            return normalizeText(root.getAttribute('data-region-page'));
        }

        function getShareUrl(){
            var canonical = document.querySelector('link[rel="canonical"]');
            var raw = canonical && canonical.href ? canonical.href : window.location.href;

            try {
                var url = new URL(raw, window.location.href);
                url.hash = '';
                return url.href;
            } catch (e) {
                return raw.split('#')[0];
            }
        }

        function getShareTitle(){
            var h1 = document.querySelector('h1.entry-title, h1.article-title, article h1, main h1');
            var title = h1 ? normalizeText(h1.textContent) : normalizeText(document.title);
            return title || '最新災害情報｜防災のまとめ';
        }

        function buildShareBlock(root){
            var block = document.createElement('aside');
            block.id = SHARE_ID;
            block.className = 'bousai-disaster-share';
            block.setAttribute('aria-label', '災害情報の共有');

            var heading = document.createElement('p');
            heading.className = 'bousai-disaster-share__title';
            heading.setAttribute('data-bousai-share-heading', '1');
            block.appendChild(heading);

            var buttons = document.createElement('div');
            buttons.className = 'bousai-disaster-share__buttons';

            var services = [
                { key:'line', label:'LINE' },
                { key:'x', label:'X' },
                { key:'threads', label:'Threads' },
                { key:'bluesky', label:'Bluesky' }
            ];

            services.forEach(function(service){
                var link = document.createElement('a');
                link.className = 'bousai-disaster-share__button bousai-disaster-share__button--' + service.key;
                link.setAttribute('data-bousai-share-service', service.key);
                link.setAttribute('target', '_blank');
                link.setAttribute('rel', 'noopener noreferrer nofollow');
                link.setAttribute('aria-label', service.label + 'で共有');

                var label = document.createElement('span');
                label.className = 'bousai-disaster-share__label';
                label.textContent = service.label;

                link.appendChild(label);
                buttons.appendChild(link);
            });

            var copy = document.createElement('button');
            copy.type = 'button';
            copy.className = 'bousai-disaster-share__button bousai-disaster-share__button--copy';
            copy.setAttribute('data-bousai-share-service', 'copy');
            copy.setAttribute('aria-label', 'リンクをコピー');

            var copyLabel = document.createElement('span');
            copyLabel.className = 'bousai-disaster-share__label';
            copyLabel.setAttribute('data-bousai-share-copy-label', '1');
            copyLabel.textContent = 'リンクをコピー';

            copy.appendChild(copyLabel);
            buttons.appendChild(copy);

            var status = document.createElement('span');
            status.className = 'bousai-disaster-share__status';
            status.setAttribute('aria-live', 'polite');
            status.setAttribute('data-bousai-share-status', '1');
            block.appendChild(buttons);
            block.appendChild(status);

            copy.addEventListener('click', function(){
                copyCurrentUrl(block, copy);
            });

            refreshShareBlock(root, block);
            return block;
        }

        function refreshShareBlock(root, block){
            if (!block) return;

            var url = getShareUrl();
            var title = getShareTitle();
            var region = getRegion(root);
            var heading = block.querySelector('[data-bousai-share-heading="1"]');

            if (heading) {
                heading.textContent = region ? 'この地域の災害情報を共有' : '全国の災害情報を共有';
            }

            var line = block.querySelector('[data-bousai-share-service="line"]');
            var x = block.querySelector('[data-bousai-share-service="x"]');
            var threads = block.querySelector('[data-bousai-share-service="threads"]');
            var bluesky = block.querySelector('[data-bousai-share-service="bluesky"]');

            if (line) {
                line.href = 'https://social-plugins.line.me/lineit/share?url=' + encodeURIComponent(url) + '&text=' + encodeURIComponent(title);
            }
            if (x) {
                x.href = 'https://x.com/intent/post?text=' + encodeURIComponent(title) + '&url=' + encodeURIComponent(url);
            }
            if (threads) {
                threads.href = 'https://www.threads.com/intent/post?text=' + encodeURIComponent(title + '\n' + url);
            }
            if (bluesky) {
                bluesky.href = 'https://bsky.app/intent/compose?text=' + encodeURIComponent(title + '\n' + url);
            }

            block.setAttribute('data-share-url', url);
            block.setAttribute('data-share-region', region || '全国');
        }

        function fallbackCopy(text){
            return new Promise(function(resolve, reject){
                var textarea = document.createElement('textarea');
                textarea.value = text;
                textarea.setAttribute('readonly', 'readonly');
                textarea.style.position = 'fixed';
                textarea.style.opacity = '0';
                textarea.style.pointerEvents = 'none';
                document.body.appendChild(textarea);
                textarea.select();

                try {
                    var ok = document.execCommand('copy');
                    document.body.removeChild(textarea);
                    ok ? resolve() : reject(new Error('copy failed'));
                } catch (e) {
                    document.body.removeChild(textarea);
                    reject(e);
                }
            });
        }

        function copyCurrentUrl(block, button){
            var url = getShareUrl();
            var status = block.querySelector('[data-bousai-share-status="1"]');
            var label = button.querySelector('[data-bousai-share-copy-label="1"]');
            var original = 'リンクをコピー';

            var job = navigator.clipboard && navigator.clipboard.writeText
                ? navigator.clipboard.writeText(url).catch(function(){ return fallbackCopy(url); })
                : fallbackCopy(url);

            job.then(function(){
                if (label) label.textContent = 'コピーしました';
                if (status) status.textContent = 'リンクをコピーしました。';
                window.setTimeout(function(){
                    if (label) label.textContent = original;
                }, 1800);
            }).catch(function(){
                if (label) label.textContent = 'コピー失敗';
                if (status) status.textContent = 'リンクをコピーできませんでした。';
                window.setTimeout(function(){
                    if (label) label.textContent = original;
                }, 2200);
            });
        }

        function placeShareBlock(){
            var root = document.querySelector('.bousai-official-info');
            if (!root) return false;

            var current = findCurrentSection(root);
            if (!current || !current.parentNode) return false;

            var slot = findPostCurrentActionsSlot(root);
            var block = document.getElementById(SHARE_ID);

            if (!block) {
                block = buildShareBlock(root);
            } else {
                refreshShareBlock(root, block);
            }

            /*
             * RC5 placement contract:
             * - stable slot exists: slot owns root-level position, SNS owns only this block
             * - CTA mounts itself at slot.firstChild; SNS stays at slot end
             * - no slot: preserve v1.4.1 fallback immediately after current section
             */
            if (slot) {
                if (block.parentNode !== slot || slot.lastElementChild !== block) {
                    slot.appendChild(block);
                }
                return true;
            }

            if (current.nextElementSibling !== block) {
                current.parentNode.insertBefore(block, current.nextSibling);
            }

            return true;
        }

        function start(){
            var root = document.querySelector('.bousai-official-info');
            if (!root) return;

            placeShareBlock();

            /*
             * layout ownerによるslot生成/再配置を待つためroot childListだけ監視する。
             * slot内に入った後はroot-level位置を奪い合わない。
             * 監視は初期描画時だけに限定する。
             */
            var observer = new MutationObserver(function(){
                placeShareBlock();
            });

            observer.observe(root, { childList:true });

            window.setTimeout(function(){
                placeShareBlock();
                observer.disconnect();
            }, 2000);
        }

        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', start, { once:true });
        } else {
            start();
        }

        window.addEventListener('pageshow', function(){
            window.setTimeout(placeShareBlock, 0);
        });
    })();
    </script>
