function bousai_weather_v1_representative_area_code( $prefecture, $areas ) {
    if ( empty( $areas ) || ! is_array( $areas ) ) return '';

    $points = bousai_weather_v1_representative_temp_points();
    $target = isset( $points[ $prefecture ] ) ? trim( (string) $points[ $prefecture ] ) : '';

    if ( $target !== '' ) {
        foreach ( $areas as $area_code => $area ) {
            $point = isset( $area['temp_point'] ) ? trim( (string) $area['temp_point'] ) : '';
            if ( $point !== '' && $point === $target ) {
                return (string) $area_code;
            }
        }
    }

    // 万一JMA側の地点名が変わっても初回を空欄にしない。
    $keys = array_keys( $areas );
    return ! empty( $keys[0] ) ? (string) $keys[0] : '';
}

function bousai_weather_v1_clean_weather_text( $text, $code = '' ) {
    $text = trim( preg_replace( '/[\x{3000}\s]+/u', ' ', (string) $text ) );
    if ( $text === '' ) {
        return bousai_weather_v1_weather_code_label( $code );
    }
    // UIでは読みやすくするため、関係語の「後」だけ「のち」に置き換える。
    $text = preg_replace( '/\s+後\s+/u', ' のち ', $text );
    return $text;
}

function bousai_weather_v1_date_key( $iso ) {
    return strlen( (string) $iso ) >= 10 ? substr( (string) $iso, 0, 10 ) : '';
}

function bousai_weather_v1_hour( $iso ) {
    return strlen( (string) $iso ) >= 13 ? (int) substr( (string) $iso, 11, 2 ) : -1;
}

function bousai_weather_v1_index_areas_by_code( $series ) {
    $out = array();
    if ( empty( $series['areas'] ) || ! is_array( $series['areas'] ) ) return $out;
    foreach ( $series['areas'] as $i => $area ) {
        $code = isset( $area['area']['code'] ) ? (string) $area['area']['code'] : '';
        if ( $code !== '' ) $out[ $code ] = array( 'index'=>$i, 'data'=>$area );
    }
    return $out;
}

function bousai_weather_v1_name_key( $name ) {
    $name = preg_replace( '/[\x{3000}\s]+/u', '', (string) $name );
    $name = str_replace( array( '地方', '北部', '南部', '東部', '西部', '中部' ), '', $name );
    return $name;
}

function bousai_weather_v1_match_weekly_area_index( $short_code, $short_name, $weekly_areas ) {
    if ( empty( $weekly_areas ) || ! is_array( $weekly_areas ) ) return null;

    foreach ( $weekly_areas as $i => $area ) {
        if ( isset( $area['area']['code'] ) && (string) $area['area']['code'] === (string) $short_code ) return $i;
    }
    if ( count( $weekly_areas ) === 1 ) return 0;

    $short_key = bousai_weather_v1_name_key( $short_name );
    $best = null;
    $best_len = 0;
    foreach ( $weekly_areas as $i => $area ) {
        $name = isset( $area['area']['name'] ) ? (string) $area['area']['name'] : '';
        $key  = bousai_weather_v1_name_key( $name );
        if ( $key === '' || $short_key === '' ) continue;
        if ( mb_strpos( $short_key, $key ) !== false || mb_strpos( $key, $short_key ) !== false ) {
            $len = mb_strlen( $key, 'UTF-8' );
            if ( $len > $best_len ) {
                $best = $i;
                $best_len = $len;
            }
        }
    }
    return $best;
}

function bousai_weather_v1_normalize_office( $office_code, $docs ) {
    if ( ! is_array( $docs ) || empty( $docs[0] ) || ! is_array( $docs[0] ) ) return false;

    $short  = $docs[0];
    $weekly = isset( $docs[1] ) && is_array( $docs[1] ) ? $docs[1] : array();
    $short_series = isset( $short['timeSeries'] ) && is_array( $short['timeSeries'] ) ? $short['timeSeries'] : array();
    if ( empty( $short_series[0]['areas'] ) ) return false;

    $weather_s = $short_series[0];
    $pop_s     = isset( $short_series[1] ) ? $short_series[1] : array();
    $temp_s    = isset( $short_series[2] ) ? $short_series[2] : array();
    $pop_by_code = bousai_weather_v1_index_areas_by_code( $pop_s );

    $weekly_series = isset( $weekly['timeSeries'] ) && is_array( $weekly['timeSeries'] ) ? $weekly['timeSeries'] : array();
    $weekly_weather = isset( $weekly_series[0] ) ? $weekly_series[0] : array();
    $weekly_temp    = isset( $weekly_series[1] ) ? $weekly_series[1] : array();
    $weekly_areas   = isset( $weekly_weather['areas'] ) && is_array( $weekly_weather['areas'] ) ? $weekly_weather['areas'] : array();

    $areas = array();

    foreach ( $weather_s['areas'] as $area_index => $weather_area ) {
        $area_code = isset( $weather_area['area']['code'] ) ? (string) $weather_area['area']['code'] : '';
        $area_name = isset( $weather_area['area']['name'] ) ? (string) $weather_area['area']['name'] : '';
        if ( $area_code === '' || $area_name === '' ) continue;

        $days = array();
        $wdefs = isset( $weather_s['timeDefines'] ) ? $weather_s['timeDefines'] : array();
        $codes = isset( $weather_area['weatherCodes'] ) ? $weather_area['weatherCodes'] : array();
        $texts = isset( $weather_area['weathers'] ) ? $weather_area['weathers'] : array();
        foreach ( $wdefs as $i => $iso ) {
            $date = bousai_weather_v1_date_key( $iso );
            if ( $date === '' ) continue;
            $code = isset( $codes[ $i ] ) ? (string) $codes[ $i ] : '';
            $text = isset( $texts[ $i ] ) ? (string) $texts[ $i ] : '';
            $days[ $date ] = array(
                'date' => $date,
                'code' => $code,
                'weather' => bousai_weather_v1_clean_weather_text( $text, $code ),
                'weather_raw' => trim( preg_replace( '/[\x{3000}\s]+/u', ' ', $text ) ),
                'pops' => array( '00'=>'', '06'=>'', '12'=>'', '18'=>'' ),
                'temp_min' => '',
                'temp_max' => '',
            );
        }

        if ( isset( $pop_by_code[ $area_code ] ) ) {
            $pops = $pop_by_code[ $area_code ]['data'];
            $pdefs = isset( $pop_s['timeDefines'] ) ? $pop_s['timeDefines'] : array();
            $vals  = isset( $pops['pops'] ) ? $pops['pops'] : array();
            foreach ( $pdefs as $i => $iso ) {
                $date = bousai_weather_v1_date_key( $iso );
                $hour = bousai_weather_v1_hour( $iso );
                if ( ! isset( $days[ $date ] ) || ! in_array( $hour, array(0,6,12,18), true ) ) continue;
                $key = str_pad( (string) $hour, 2, '0', STR_PAD_LEFT );
                $days[ $date ]['pops'][ $key ] = isset( $vals[ $i ] ) ? (string) $vals[ $i ] : '';
            }
        }

        // 気温予報地点は短期予報の一次細分区域と同じ並びで提供されるため、同じindexを対応させる。
        if ( ! empty( $temp_s['areas'][ $area_index ] ) ) {
            $temp_area = $temp_s['areas'][ $area_index ];
            $tdefs = isset( $temp_s['timeDefines'] ) ? $temp_s['timeDefines'] : array();
            $vals  = isset( $temp_area['temps'] ) ? $temp_area['temps'] : array();
            foreach ( $tdefs as $i => $iso ) {
                $date = bousai_weather_v1_date_key( $iso );
                $hour = bousai_weather_v1_hour( $iso );
                if ( ! isset( $days[ $date ] ) || ! isset( $vals[ $i ] ) || $vals[ $i ] === '' ) continue;
                if ( $hour === 0 ) $days[ $date ]['temp_min'] = (string) $vals[ $i ];
                if ( $hour === 9 ) $days[ $date ]['temp_max'] = (string) $vals[ $i ];
            }
        }

        $weekly_days = array();
        $week_index = bousai_weather_v1_match_weekly_area_index( $area_code, $area_name, $weekly_areas );
        if ( $week_index !== null && isset( $weekly_areas[ $week_index ] ) ) {
            $wa = $weekly_areas[ $week_index ];
            $wdefs = isset( $weekly_weather['timeDefines'] ) ? $weekly_weather['timeDefines'] : array();
            $wcodes = isset( $wa['weatherCodes'] ) ? $wa['weatherCodes'] : array();
            $wpops  = isset( $wa['pops'] ) ? $wa['pops'] : array();
            $rels   = isset( $wa['reliabilities'] ) ? $wa['reliabilities'] : array();
            $temp_area = isset( $weekly_temp['areas'][ $week_index ] ) ? $weekly_temp['areas'][ $week_index ] : array();
            $mins = isset( $temp_area['tempsMin'] ) ? $temp_area['tempsMin'] : array();
            $maxs = isset( $temp_area['tempsMax'] ) ? $temp_area['tempsMax'] : array();
            foreach ( $wdefs as $i => $iso ) {
                $date = bousai_weather_v1_date_key( $iso );
                if ( $date === '' ) continue;
                $code = isset( $wcodes[ $i ] ) ? (string) $wcodes[ $i ] : '';
                $weekly_days[] = array(
                    'date' => $date,
                    'code' => $code,
                    'weather' => bousai_weather_v1_weather_code_label( $code ),
                    'pop' => isset( $wpops[ $i ] ) ? (string) $wpops[ $i ] : '',
                    'reliability' => isset( $rels[ $i ] ) ? (string) $rels[ $i ] : '',
                    'temp_min' => isset( $mins[ $i ] ) ? (string) $mins[ $i ] : '',
                    'temp_max' => isset( $maxs[ $i ] ) ? (string) $maxs[ $i ] : '',
                );
            }
        }

        $areas[ $area_code ] = array(
            'code' => $area_code,
            'name' => $area_name,
            'office_code' => (string) $office_code,
            'temp_point' => isset( $temp_s['areas'][ $area_index ]['area']['name'] ) ? (string) $temp_s['areas'][ $area_index ]['area']['name'] : '',
            'days' => array_values( $days ),
            'weekly' => $weekly_days,
        );
    }

    return array(
        'office_code' => (string) $office_code,
        'publishing_office' => isset( $short['publishingOffice'] ) ? (string) $short['publishingOffice'] : '気象庁',
        'short_report_at' => isset( $short['reportDatetime'] ) ? (string) $short['reportDatetime'] : '',
        'weekly_report_at' => isset( $weekly['reportDatetime'] ) ? (string) $weekly['reportDatetime'] : '',
        'fetched_at' => time(),
        'areas' => $areas,
    );
}

function bousai_weather_v1_fetch_office( $office_code ) {
    $url = 'https://www.jma.go.jp/bosai/forecast/data/forecast/' . rawurlencode( $office_code ) . '.json';
    $response = wp_safe_remote_get( $url, array(
        'timeout' => 4,
        'redirection' => 1,
        'user-agent' => 'Bousai-Matome/1.0; ' . home_url( '/' ),
        'headers' => array( 'Accept' => 'application/json' ),
    ) );
    if ( is_wp_error( $response ) ) return false;
    if ( 200 !== (int) wp_remote_retrieve_response_code( $response ) ) return false;
    $body = wp_remote_retrieve_body( $response );
    if ( $body === '' ) return false;
    $docs = json_decode( $body, true );
    if ( JSON_ERROR_NONE !== json_last_error() || ! is_array( $docs ) ) return false;
    return bousai_weather_v1_normalize_office( $office_code, $docs );
}

function bousai_weather_v1_start_refresh() {
    $build = get_option( BOUSAI_WEATHER_V1_BUILD_OPTION, array() );
    if ( is_array( $build ) && ! empty( $build['remaining'] ) && ! empty( $build['started_at'] ) && ( time() - (int) $build['started_at'] ) < 1800 ) {
        return;
    }

    $active = get_option( BOUSAI_WEATHER_V1_SNAPSHOT_OPTION, array() );
    $existing = is_array( $active ) && ! empty( $active['offices'] ) && is_array( $active['offices'] ) ? $active['offices'] : array();
    $build = array(
        'version' => BOUSAI_WEATHER_V1_VERSION,
        'started_at' => time(),
        'remaining' => bousai_weather_v1_all_office_codes(),
        'offices' => $existing,
        'failed' => array(),
    );
    update_option( BOUSAI_WEATHER_V1_BUILD_OPTION, $build, false );

    if ( ! wp_next_scheduled( 'bousai_weather_v1_process_batch_event' ) ) {
        wp_schedule_single_event( time() + 2, 'bousai_weather_v1_process_batch_event' );
    }
}

function bousai_weather_v1_process_batch() {
    if ( get_transient( BOUSAI_WEATHER_V1_LOCK ) ) return;
    set_transient( BOUSAI_WEATHER_V1_LOCK, 1, 90 );

    try {
        $build = get_option( BOUSAI_WEATHER_V1_BUILD_OPTION, array() );
        if ( ! is_array( $build ) || empty( $build['remaining'] ) || ! is_array( $build['remaining'] ) ) return;

        $batch = array_splice( $build['remaining'], 0, 6 );
        foreach ( $batch as $office_code ) {
            $normalized = bousai_weather_v1_fetch_office( $office_code );
            if ( $normalized ) {
                $build['offices'][ $office_code ] = $normalized;
                unset( $build['failed'][ $office_code ] );
            } else {
                $build['failed'][ $office_code ] = time();
            }
        }
        update_option( BOUSAI_WEATHER_V1_BUILD_OPTION, $build, false );

        // 取得済みofficeは全国一巡の完了を待たず表示用スナップショットへ反映する。
        $partial = get_option( BOUSAI_WEATHER_V1_SNAPSHOT_OPTION, array() );
        if ( ! is_array( $partial ) ) $partial = array();
        $partial['version'] = BOUSAI_WEATHER_V1_VERSION;
        $partial['offices'] = isset( $build['offices'] ) && is_array( $build['offices'] ) ? $build['offices'] : array();
        $partial['failed'] = isset( $build['failed'] ) && is_array( $build['failed'] ) ? $build['failed'] : array();
        if ( ! isset( $partial['updated_at'] ) ) $partial['updated_at'] = 0;
        update_option( BOUSAI_WEATHER_V1_SNAPSHOT_OPTION, $partial, false );

        if ( ! empty( $build['remaining'] ) ) {
            wp_schedule_single_event( time() + 12, 'bousai_weather_v1_process_batch_event' );
            return;
        }

        $snapshot = array(
            'version' => BOUSAI_WEATHER_V1_VERSION,
            'updated_at' => time(),
            'offices' => isset( $build['offices'] ) && is_array( $build['offices'] ) ? $build['offices'] : array(),
            'failed' => isset( $build['failed'] ) && is_array( $build['failed'] ) ? $build['failed'] : array(),
        );
        update_option( BOUSAI_WEATHER_V1_SNAPSHOT_OPTION, $snapshot, false );
        delete_option( BOUSAI_WEATHER_V1_BUILD_OPTION );
    } finally {
        delete_transient( BOUSAI_WEATHER_V1_LOCK );
    }
}
add_action( 'bousai_weather_v1_process_batch_event', 'bousai_weather_v1_process_batch' );

function bousai_weather_v1_current_prefecture() {
    if ( is_admin() ) return '';

    // 全国 /disaster-info/ には絶対に出さない。
    // 仮想都道府県URL /disaster-info/{slug}/ のときだけ対象にする。
    $uri  = isset( $_SERVER['REQUEST_URI'] ) ? (string) $_SERVER['REQUEST_URI'] : '';
    $path = wp_parse_url( $uri, PHP_URL_PATH );
    $path = is_string( $path ) ? trim( $path, '/' ) : '';

    if ( ! preg_match( '#^disaster-info/([a-z0-9_-]+)$#', $path, $m ) ) {
        return '';
    }

    $slug = sanitize_title( (string) $m[1] );

    if ( function_exists( 'bousai_matome_region_name_from_slug' ) ) {
        $region = trim( (string) bousai_matome_region_name_from_slug( $slug ) );
        if ( $region !== '' ) return $region;
    }

    return '';
}

function bousai_weather_v1_payload_for_prefecture( $prefecture ) {
    $map = bousai_weather_v1_prefecture_offices();
    if ( empty( $map[ $prefecture ] ) ) return false;

    $snapshot = get_option( BOUSAI_WEATHER_V1_SNAPSHOT_OPTION, array() );
    if ( ! is_array( $snapshot ) || empty( $snapshot['offices'] ) || ! is_array( $snapshot['offices'] ) ) return false;

    $areas = array();
    $latest_fetch = 0;
    foreach ( $map[ $prefecture ] as $office_code ) {
        if ( empty( $snapshot['offices'][ $office_code ] ) || ! is_array( $snapshot['offices'][ $office_code ] ) ) continue;
        $office = $snapshot['offices'][ $office_code ];
        $fetched = isset( $office['fetched_at'] ) ? (int) $office['fetched_at'] : 0;
        if ( $fetched > $latest_fetch ) $latest_fetch = $fetched;

        // 一部地域だけ取得に失敗した場合、その地域の古い予報を他地域の新しい取得時刻で隠さない。
        // 各予報区ごとに6時間を超えた保存データは表示対象から外す。
        if ( ! $fetched || ( time() - $fetched ) > 21600 ) continue;
        if ( empty( $office['areas'] ) || ! is_array( $office['areas'] ) ) continue;
        foreach ( $office['areas'] as $area_code => $area ) {
            $area['short_report_at'] = isset( $office['short_report_at'] ) ? (string) $office['short_report_at'] : '';
            $area['weekly_report_at'] = isset( $office['weekly_report_at'] ) ? (string) $office['weekly_report_at'] : '';
            $area['fetched_at'] = $fetched;
            $areas[ $area_code ] = $area;
        }
    }
    if ( empty( $areas ) ) return false;

    // 全予報区が古い場合は何も表示しない。上で地域単位の鮮度判定も済ませている。
    if ( empty( $areas ) || ! $latest_fetch ) return false;

    $default_area_code = bousai_weather_v1_representative_area_code( $prefecture, $areas );

    return array(
        'version' => BOUSAI_WEATHER_V1_VERSION,
        'prefecture' => $prefecture,
        'generated_at' => time(),
        'today_jst' => wp_date( 'Y-m-d', time(), new DateTimeZone( 'Asia/Tokyo' ) ),
        'default_area_code' => $default_area_code,
        'areas' => array_values( $areas ),
    );
}

    return bousai_weather_v1_current_prefecture() !== '';
}

 var positions=[]; tests.forEach(function(t){ var i=s.search(t[1]); if(i>=0)positions.push({type:t[0],i:i}); });
 positions.sort(function(a,b){return a.i-b.i;}); positions.forEach(function(x){ if(types.indexOf(x.type)<0)types.push(x.type); });
 if(!types.length){ var f=String(code||'').charAt(0); types=[f==='1'?'sun':f==='2'?'cloud':f==='3'?'rain':f==='4'?'snow':'cloud']; }
 // 雷は補助現象として末尾へ。主天気2種までに抑える。
 if(types.length>2) types=types.slice(0,2);
 var rel=''; if(types.length>1){ if(/のち|後/.test(s))rel='arrow'; else if(/時々|一時/.test(s))rel='slash'; }
 return {types:types,relation:rel,heavy:/大雨|激しく|非常に激しく|猛烈/.test(s)};
}

function place(){
 var slot=root.querySelector(':scope > [data-bousai-slot="post-current-actions"]');
 if(!slot)return false;
 // Weather keeps its original slot; it no longer depends on SNS or CTA mounting.
 // Move only the weather node, never the slot, CTA, SNS, or source sections.
 var cta=slot.querySelector(':scope > [data-bcc-surface="disaster_info_inline"]');
 var weather=document.getElementById(ID)||box;
 if(cta){
   if(weather.parentNode!==slot || weather.nextElementSibling!==cta)slot.insertBefore(weather,cta);
 }else if(weather.parentNode!==slot || slot.firstElementChild!==weather){
   slot.insertBefore(weather,slot.firstChild);
 }
 return weather.parentNode===slot;
}
